cmake_minimum_required(VERSION 3.25)
project(OpenCrownMobileMetal LANGUAGES C CXX OBJC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DCP_SDL_SOURCE "" CACHE PATH "Pinned SDL 3.4.18 source directory")
if(NOT EXISTS "${DCP_SDL_SOURCE}/include/SDL3/SDL.h")
  message(FATAL_ERROR "Pass -DDCP_SDL_SOURCE=<cached SDL 3.4.18 source>")
endif()
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
set(SDL_CAMERA OFF CACHE BOOL "" FORCE)
add_subdirectory("${DCP_SDL_SOURCE}" sdl)
get_filename_component(DCP_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../linux-port" ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_program(DCP_SPIRV_CROSS spirv-cross REQUIRED)
# project() may normalize a short SDK name to an absolute iPhoneSimulator SDK
# path. Match case-insensitively so simulator code never embeds device libraries.
string(TOLOWER "${CMAKE_OSX_SYSROOT}" DCP_OSX_SYSROOT_LOWER)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  if(DCP_OSX_SYSROOT_LOWER MATCHES "iphonesimulator")
    set(DCP_METAL_SDK iphonesimulator)
  else()
    set(DCP_METAL_SDK iphoneos)
  endif()
else()
  set(DCP_METAL_SDK macosx)
endif()
if(CMAKE_OSX_DEPLOYMENT_TARGET)
  set(DCP_MINIMUM_OS "${CMAKE_OSX_DEPLOYMENT_TARGET}")
elseif(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(DCP_MINIMUM_OS 18.0)
else()
  set(DCP_MINIMUM_OS 15.0)
endif()
file(GLOB DCP_SHADERS CONFIGURE_DEPENDS "${DCP_SOURCE}/src/render/shaders/generated/*.spv.h")
set(DCP_METAL_HEADER "${CMAKE_CURRENT_BINARY_DIR}/shaders/metal_shaders.h")
add_custom_command(OUTPUT "${DCP_METAL_HEADER}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate-metal.py"
    --source "${DCP_SOURCE}/src/render/shaders/generated" --output "${DCP_METAL_HEADER}"
    --sdk "${DCP_METAL_SDK}" --minimum-os "${DCP_MINIMUM_OS}" --spirv-cross "${DCP_SPIRV_CROSS}"
  DEPENDS ${DCP_SHADERS} "${CMAKE_CURRENT_SOURCE_DIR}/generate-metal.py" VERBATIM)
add_library(opencrown-mobile-metal STATIC metal-device.cpp bc-textures.cpp "${DCP_SOURCE}/src/render/gpu_device.cpp" "${DCP_METAL_HEADER}")
target_include_directories(opencrown-mobile-metal PUBLIC "${DCP_SOURCE}/src" "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/shaders")
target_compile_definitions(opencrown-mobile-metal PUBLIC DCP_HAVE_SDL_GPU DCP_AOT)
target_link_libraries(opencrown-mobile-metal PUBLIC SDL3::SDL3-static)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_executable(metal-fixture fixture.cpp "${DCP_SOURCE}/src/render/png.cpp")
  target_link_libraries(metal-fixture PRIVATE opencrown-mobile-metal)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_executable(capture-fixture capture-fixture.cpp "${DCP_SOURCE}/src/render/capture.cpp")
  target_include_directories(capture-fixture PRIVATE "${DCP_SOURCE}/src")
  target_compile_definitions(capture-fixture PRIVATE DCP_AOT)
endif()

set(DCP_FREETYPE_SOURCE "" CACHE PATH "Cached FreeType source directory")
if(EXISTS "${DCP_FREETYPE_SOURCE}/CMakeLists.txt")
  set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
  set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
  set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
  set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
  set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
  add_subdirectory("${DCP_FREETYPE_SOURCE}" freetype)
  add_library(opencrown-mobile-replay STATIC replay.cpp
    "${DCP_SOURCE}/src/render/backend.cpp" "${DCP_SOURCE}/src/render/capture.cpp"
    "${DCP_SOURCE}/src/render/prompts.cpp" "${DCP_SOURCE}/src/render/png.cpp"
    "${DCP_SOURCE}/src/formats/ftx.cpp" "${DCP_SOURCE}/src/formats/gnm_detile.cpp")
  target_include_directories(opencrown-mobile-replay PUBLIC "${DCP_SOURCE}/src")
  target_compile_definitions(opencrown-mobile-replay PUBLIC DCP_AOT)
  target_link_libraries(opencrown-mobile-replay PUBLIC opencrown-mobile-metal freetype)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    add_executable(texture-fixture texture-fixture.cpp)
    target_link_libraries(texture-fixture PRIVATE opencrown-mobile-replay)
    add_executable(texture-snapshot-fixture texture-snapshot-fixture.cpp)
    target_link_libraries(texture-snapshot-fixture PRIVATE opencrown-mobile-replay)
    target_compile_options(texture-snapshot-fixture PRIVATE -UNDEBUG)
    add_executable(replay-fixture replay-fixture.cpp)
    target_link_libraries(replay-fixture PRIVATE opencrown-mobile-replay)
    target_compile_options(replay-fixture PRIVATE -UNDEBUG)
  endif()
endif()
if(TARGET capture-fixture)
  target_compile_options(capture-fixture PRIVATE -UNDEBUG)
endif()
if(TARGET texture-fixture)
  target_compile_options(texture-fixture PRIVATE -UNDEBUG)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_executable(bc-fixture bc-fixture.cpp)
  target_link_libraries(bc-fixture PRIVATE opencrown-mobile-metal)
  target_compile_options(bc-fixture PRIVATE -UNDEBUG)
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/ThirdPartyNotices")
configure_file("${DCP_SDL_SOURCE}/LICENSE.txt" "${CMAKE_CURRENT_BINARY_DIR}/ThirdPartyNotices/SDL3-LICENSE.txt" COPYONLY)
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/third_party/bcdec-LICENSE" "${CMAKE_CURRENT_BINARY_DIR}/ThirdPartyNotices/bcdec-LICENSE" COPYONLY)
if(EXISTS "${DCP_FREETYPE_SOURCE}/docs/FTL.TXT")
  configure_file("${DCP_FREETYPE_SOURCE}/docs/FTL.TXT" "${CMAKE_CURRENT_BINARY_DIR}/ThirdPartyNotices/FreeType-FTL.txt" COPYONLY)
endif()
